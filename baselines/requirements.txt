# numpy 2 is fine on Apple Silicon/Linux, but the last macOS x86_64 TensorFlow
# wheel (tensorflow-cpu 2.16.2) caps numpy at <2.0 — pinning >=2.0 makes the
# install unsolvable on Intel Macs. A floor of 1.26 lets pip pick 2.x where the
# platform allows it and 1.26 where it doesn't. The code uses no numpy-2-only API.
numpy>=1.26,<3.0
pandas>=2.0
scipy>=1.11
scikit-learn>=1.3
matplotlib>=3.7
tensorflow-cpu>=2.15,<2.20

# TF 2.16.2 still does `import distutils`, which Python 3.12 removed from the
# stdlib (PEP 632). setuptools used to backfill it, but setuptools 81 dropped
# that shim — so on 3.12 the import fails with ModuleNotFoundError: distutils
# unless setuptools is held below 81. Harmless on Python <=3.11 and on newer TF.
setuptools<81
